# ========================================== # 1. ส่วนคำสั่งติดตั้งระบบ (สำหรับรันใน Terminal) # ========================================== # ก่อนรันโค้ดนี้ ให้พิมพ์คำสั่งเหล่านี้ใน Terminal ของ GCP เพื่อเตรียมระบบ: # sudo apt update && sudo apt install -y adb android-tools-adb python3-pip # pip install fastapi uvicorn websockets pure-python-adb # adb start-server # python3 app.py # ========================================== import os import json import base64 import asyncio from fastapi import FastAPI, WebSocket from fastapi.responses import HTMLResponse from ppadb.client import Client as AdbClient app = FastAPI() # ---------------------------------------------------- # ⚙️ CONFIGURATION (ตั้งค่าการเชื่อมต่อ) # ---------------------------------------------------- # IP ภายในเครื่อง Server (หาก Android Emulator รันอยู่บนเครื่องเดียวกัน) CLOUD_PHONE_IP = "127.0.0.1" CLOUD_PHONE_PORT = 5555 SCREEN_WIDTH = 1080 # ความละเอียดหน้าจอแนวนอนของมือถือคลาวด์ SCREEN_HEIGHT = 1920 # ความละเอียดหน้าจอแนวตั้งของมือถือคลาวด์ def connect_adb_device(): """ฟังก์ชันเชื่อมต่อไปยังเครื่อง Cloud Phone ผ่าน ADB""" client = AdbClient(host="127.0.0.1", port=5037) try: client.remote_connect(CLOUD_PHONE_IP, CLOUD_PHONE_PORT) device = client.device(f"{CLOUD_PHONE_IP}:{CLOUD_PHONE_PORT}") return device except Exception as e: print(f"🚨 ADB Connection Error: {e}") return None # ---------------------------------------------------- # 📱 FRONTEND UI (หน้าจอควบคุม - รวมอยู่ในนี้เลย) # ---------------------------------------------------- HTML_CONTENT = """ GCP Cloud Phone Controller

☁️ GCP Cloud Phone Panel

สถานะระบบ: กำลังเชื่อมต่อสัญญาณ...
""" @app.get("/", response_class=HTMLResponse) async def get_index(): """ส่งหน้าจอควบคุมหลักให้ผู้ใช้งาน""" return HTML_CONTENT # ---------------------------------------------------- # ⚙️ BACKEND CORE (ระบบประมวลผลคำสั่งและจัดการวิดีโอ) # ---------------------------------------------------- @app.websocket("/ws/stream") async def websocket_stream_endpoint(websocket: WebSocket): await websocket.accept() device = connect_adb_device() if not device: print("🚨 คำเตือน: ยังไม่ได้เปิดเครื่องจำลอง Android หรือต่อ ADB ไม่ติด") # ระบบส่งภาพหน้าจอ (สตรีมมิงออกจากเซิร์ฟเวอร์ GCP) async def capture_loop(): try: while True: if device: screenshot_bytes = device.screencap() base64_image = base64.b64encode(screenshot_bytes).decode('utf-8') await websocket.send_text(f"data:image/png;base64,{base64_image}") await asyncio.sleep(0.07) # สตรีมเฟรมเรตคงที่ except Exception as e: print(f"การสตรีมภาพหยุดทำงาน: {e}") # ระบบรับคำสั่งคลิก/กดปุ่ม จากหน้าเว็บไปกดบนมือถือจริง async def command_loop(): try: while True: data = await websocket.receive_text() command = json.loads(data) if not device: continue if command["action"] == "tap": real_x = int(command["x_percent"] * SCREEN_WIDTH) real_y = int(command["y_percent"] * SCREEN_HEIGHT) device.shell(f"input tap {real_x} {real_y}") elif command["action"] == "keyevent": key_code = command["code"] device.shell(f"input keyevent {key_code}") except Exception as e: print(f"ระบบสั่งการหยุดทำงาน: {e}") # รันทั้งสองระบบพร้อมกันแบบคู่ขนาน await asyncio.gather(capture_loop(), command_loop()) if __name__ == "__main__": import uvicorn # เปิดบริการที่พอร์ต 80 เพื่อให้พิมพ์แค่ IP ของ Google Cloud บนเว็บแล้วเข้าได้ทันที uvicorn.run(app, host="0.0.0.0", port=80)
