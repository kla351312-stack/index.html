// ==UserScript== // @name BluezyGPT เปลี่ยนชื่อและโลโก้เป็น KLA // @namespace http://tampermonkey.net // @version 1.1 // @description เปลี่ยนข้อความ BluezyGPT เป็น KLA และเปลี่ยนรูปโลโก้หน้าเว็บ // @author You // @match https://bluezygpt.space* // @grant none // @run-at document-end // ==/UserScript== (function() { 'use strict'; // 1. ฟังก์ชันสำหรับเปลี่ยนข้อความบนหน้าจอ function renameText() { const elements = document.querySelectorAll('body *:not(script):not(style)'); elements.forEach(element => { if (element.children.length === 0 && element.textContent.includes('BluezyGPT')) { element.textContent = element.textContent.replace(/BluezyGPT/g, 'KLA'); } }); } // 2. ฟังก์ชันสำหรับเปลี่ยนรูปภาพโลโก้หน้าเว็บ function replaceLogo() { // ใส่ลิงก์ URL รูปภาพโลโก้ KLA ของคุณในช่องเครื่องหมายคำพูดด้านล่างนี้ const newLogoUrl = "วาง_ลิงก์_URL_รูปภาพโลโก้ตรงนี้"; // ค้นหารูปภาพโลโก้เดิมบนหน้าเว็บ (สแกนหาแท็ก img ทั้งหมด) const images = document.querySelectorAll('img'); images.forEach(img => { // ดักจับจากชื่อไฟล์เดิม หรือ คลาสที่เกี่ยวกับ logo (ปรับแต่งเพิ่มเติมได้ตามโครงสร้างหน้าเว็บจริง) if (img.src.includes('logo') || img.alt.includes('logo') || img.className.includes('logo')) { img.src = newLogoUrl; } }); } // รวมฟังก์ชันการทำงานทั้งหมด function runKLAChanges() { renameText(); replaceLogo(); } // สั่งให้รันครั้งแรกเมื่อโหลดหน้าเว็บ runKLAChanges(); // ติดตามตรวจสอบการเปลี่ยนแปลงของหน้าเว็บแบบ Dynamic const observer = new MutationObserver(runKLAChanges); observer.observe(document.body, { childList: true, subtree: true }); })();
